async function loadMessages() { if (!currentChatPartnerId) return; const container = document.getElementById('chatMessages'); const cleanUserId = String(currentUserId).trim(); const cleanPartnerId = String(currentChatPartnerId).trim(); const { data, error } = await supabaseClient .from('messages') .select('*') .or(`and(sender_id.eq.${cleanUserId},receiver_id.eq.${cleanPartnerId}),and(sender_id.eq.${cleanPartnerId},receiver_id.eq.${cleanUserId})`) .order('created_at', { ascending: true }); if (error) { container.innerHTML = `
โหลดข้อความไม่สำเร็จ
`; return; } if (!data || data.length === 0) { container.innerHTML = `
ยังไม่มีข้อความ ทักทายคนแรกเลย! ✨
`; return; } container.innerHTML = data.map(msg => { const isMe = String(msg.sender_id).trim() === cleanUserId; const timeStr = msg.created_at ? new Date(msg.created_at).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }) : ''; return `
${msg.image_url ? `` : ''} ${msg.content ? `

${msg.content}

` : ''}
${timeStr ? `${timeStr}` : ''}
`; }).join(''); container.scrollTop = container.scrollHeight; } async function sendMessage() { const input = document.getElementById('chatInput'); const content = input.value.trim(); let imageUrl = null; const cleanUserId = String(currentUserId).trim(); const cleanPartnerId = String(currentChatPartnerId).trim(); if ((!content && !selectedFile) || !cleanPartnerId) return; if (selectedFile) { const fileName = `${Date.now()}_${selectedFile.name}`; const { data: imgData, error: imgError } = await supabaseClient .storage .from('chat-images') .upload(fileName, selectedFile); if (!imgError && imgData) { const { data: publicUrlData } = supabaseClient .storage .from('chat-images') .getPublicUrl(fileName); imageUrl = publicUrlData.publicUrl; } } // ส่งเฉพาะ sender_id, receiver_id และ content ส่วน id กับ created_at ให้ Supabase สร้างให้อัตโนมัติ const insertData = { sender_id: cleanUserId, receiver_id: cleanPartnerId, content: content }; if (imageUrl) { insertData.image_url = imageUrl; } const { error } = await supabaseClient .from('messages') .insert([insertData]); if (!error) { input.value = ''; clearImage(); loadMessages(); } else { alert('ส่งข้อความไม่สำเร็จ: ' + error.message); } }
